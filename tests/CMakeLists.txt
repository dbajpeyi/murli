cmake_minimum_required(VERSION 3.22)
project(murli_engine_test CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(murli_engine_test engine_test.cpp)
if(NOT MSVC)
  target_compile_options(murli_engine_test PRIVATE -Wall -Wextra -Wshadow)
endif()
